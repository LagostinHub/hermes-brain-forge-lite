## Config Paths (Windows)

```yaml
# Davinci Resolve
davinci-resolve:
  command: python
  args: [-m, davinci_mcp]
  cwd: C:/AI/tools/wasserman-suite/unofficial-davinci-mcp

# Kan
kan:
  command: %USERPROFILE%\AppData\Local\hermes\node\node.exe
  args: [C:/AI/kan-filmes/packages/mcp/dist/index.js]
  env:
    KAN_BASE_URL: https://lagos-kanban.vercel.app
    KAN_API_TOKEN: ${env:KAN_API_TOKEN}

# ScriptBreak / Cork-Board / Blockout (stdio node)
scriptbreak:
  command: node
  args: [C:/AI/tools/scriptbreak/mcp/scriptbreak-mcp.mjs]

cork-board:
  command: node
  args: [C:/AI/tools/wasserman-suite/cork-board/mcp/cork-board-mcp.mjs]

blockout:
  command: node
  args: [C:/AI/tools/wasserman-suite/blockout/mcp/blockout-mcp.mjs]

openshorts:
  command: python
  args: [C:/AI/openshorts/mcp_stdio.py]
  timeout: 60
  env:
    GEMINI_API_KEY: ${env:GEMINI_API_KEY}
```
